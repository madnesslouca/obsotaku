cmake_minimum_required(VERSION 3.20)
project(studio-wasapi-capture LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This fixture must be built for native Windows")
endif()
add_executable(studio-wasapi-loopback wasapi-loopback.cpp)
target_compile_features(studio-wasapi-loopback PRIVATE cxx_std_17)
target_compile_definitions(studio-wasapi-loopback PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(studio-wasapi-loopback PRIVATE ole32 uuid propsys)
if(MSVC)
  target_compile_options(studio-wasapi-loopback PRIVATE /W4 /EHsc)
endif()
