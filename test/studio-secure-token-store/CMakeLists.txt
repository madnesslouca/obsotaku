cmake_minimum_required(VERSION 3.28)
project(studio-secure-token-store LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Threads REQUIRED)
set(STORE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../frontend/utility")
add_executable(studio-secure-token-store-check check.cpp "${STORE_ROOT}/SecureTokenStore.cpp")
target_include_directories(studio-secure-token-store-check PRIVATE "${STORE_ROOT}")
target_link_libraries(studio-secure-token-store-check PRIVATE Threads::Threads)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(Libsecret REQUIRED IMPORTED_TARGET libsecret-1>=0.18)
  target_sources(studio-secure-token-store-check PRIVATE "${STORE_ROOT}/SecureTokenStore_Linux.cpp")
  target_link_libraries(studio-secure-token-store-check PRIVATE PkgConfig::Libsecret)
elseif(WIN32)
  target_sources(studio-secure-token-store-check PRIVATE "${STORE_ROOT}/SecureTokenStore_Windows.cpp")
  target_link_libraries(studio-secure-token-store-check PRIVATE Advapi32)
  target_compile_definitions(studio-secure-token-store-check PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(studio-secure-token-store-check PRIVATE /utf-8)
else()
  message(FATAL_ERROR "This test requires Secret Service or Windows Credential Manager")
endif()
