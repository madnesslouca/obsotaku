cmake_minimum_required(VERSION 3.28)
project(studio_mpegts_start_failure_checks LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
foreach(input SDK_SOURCE SDK_BUILD CORE_LIBRARY AFTER_SOURCE_ROOT)
  if(NOT IS_ABSOLUTE "${${input}}" OR NOT EXISTS "${${input}}")
    message(FATAL_ERROR "Set ${input} to an absolute existing matching input")
  endif()
endforeach()
set(CONTROL_SOURCE_ROOT "" CACHE PATH "Optional preceding disposal revision with matching ownership layout")
if(CONTROL_SOURCE_ROOT AND (NOT IS_ABSOLUTE "${CONTROL_SOURCE_ROOT}" OR NOT EXISTS "${CONTROL_SOURCE_ROOT}"))
  message(FATAL_ERROR "CONTROL_SOURCE_ROOT must be absolute and exist")
endif()
list(APPEND CMAKE_MODULE_PATH "${SDK_SOURCE}/cmake/finders")
find_package(libobs CONFIG REQUIRED)
find_package(Threads REQUIRED)
find_package(FFmpeg REQUIRED COMPONENTS avformat avcodec avutil swscale)
find_package(Librist REQUIRED)
find_package(Libsrt REQUIRED)
set(input_manifest "{}")
function(record_input key path)
  file(SHA256 "${path}" value)
  string(JSON input_manifest SET "${input_manifest}" "${key}" "\"${value}\"")
  set(input_manifest "${input_manifest}" PARENT_SCOPE)
endfunction()
record_input(core_link_library "${CORE_LIBRARY}")
foreach(name check.c stop-overlap.c slow-close.c opaque-close.c preflight.c)
  record_input("fixture/${name}" "${CMAKE_CURRENT_SOURCE_DIR}/${name}")
endforeach()
foreach(name libobs/obs.h plugins/obs-ffmpeg/obs-ffmpeg-formats.h plugins/obs-ffmpeg/obs-ffmpeg-compat.h)
  record_input("sdk/${name}" "${SDK_SOURCE}/${name}")
endforeach()
function(configure_check target)
  target_include_directories(${target} BEFORE PRIVATE "${SDK_SOURCE}/libobs" "${SDK_SOURCE}/deps/libcaption" "${SDK_BUILD}/config" "${SDK_BUILD}/libobs" $<TARGET_PROPERTY:OBS::libobs,INTERFACE_INCLUDE_DIRECTORIES>)
  target_compile_definitions(${target} PRIVATE HAVE_OBSCONFIG_H $<TARGET_PROPERTY:OBS::libobs,INTERFACE_COMPILE_DEFINITIONS>)
  target_link_libraries(${target} PRIVATE "${CORE_LIBRARY}" $<TARGET_PROPERTY:OBS::libobs,INTERFACE_LINK_LIBRARIES> Threads::Threads FFmpeg::avformat FFmpeg::avcodec FFmpeg::avutil FFmpeg::swscale Librist::Librist Libsrt::Libsrt $<$<PLATFORM_ID:Windows>:ws2_32> $<$<NOT:$<PLATFORM_ID:Windows>>:m>)
  if(MSVC)
    target_compile_options(${target} PRIVATE /utf-8 /Gy /Gw)
    target_compile_definitions(${target} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_link_options(${target} PRIVATE /OPT:REF)
  else()
    target_compile_options(${target} PRIVATE -ffunction-sections -fdata-sections -O1)
    target_link_options(${target} PRIVATE -Wl,--gc-sections)
  endif()
endfunction()
add_executable(preflight preflight.c)
configure_check(preflight)
function(add_variant label root)
  foreach(name obs-ffmpeg-mpegts.c obs-ffmpeg-output.h)
    record_input("${label}/${name}" "${root}/plugins/obs-ffmpeg/${name}")
  endforeach()
  set(input_manifest "${input_manifest}" PARENT_SCOPE)
  foreach(family check overlap slow opaque)
    if(family STREQUAL check)
      set(body check.c)
    elseif(family STREQUAL overlap)
      set(body stop-overlap.c)
    elseif(family STREQUAL slow)
      set(body slow-close.c)
    else()
      set(body opaque-close.c)
    endif()
    add_executable(${family}-${label} "${body}")
    configure_check(${family}-${label})
    target_include_directories(${family}-${label} BEFORE PRIVATE "${root}/plugins/obs-ffmpeg" "${SDK_SOURCE}/plugins/obs-ffmpeg")
    target_compile_definitions(${family}-${label} PRIVATE NEW_MPEGTS_OUTPUT TEST_CUSTODY TEST_MPEGTS_SOURCE="${root}/plugins/obs-ffmpeg/obs-ffmpeg-mpegts.c")
  endforeach()
endfunction()
add_variant(after "${AFTER_SOURCE_ROOT}")
if(CONTROL_SOURCE_ROOT)
  add_variant(control "${CONTROL_SOURCE_ROOT}")
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/sdk-manifest.json" "${input_manifest}\n")
