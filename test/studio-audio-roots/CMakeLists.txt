cmake_minimum_required(VERSION 3.28)
project(studio-audio-roots-check LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/finders")
find_package(libobs REQUIRED)
find_package(Threads REQUIRED)
set(STUDIO_AUDIO_TEST_CORE "" CACHE FILEPATH "Exact private relinked core")
if(NOT EXISTS "${STUDIO_AUDIO_TEST_CORE}")
 message(FATAL_ERROR "Use a private test core, never modify installed libobs")
endif()
add_executable(studio-audio-roots-check check.cpp)
target_include_directories(studio-audio-roots-check PRIVATE $<TARGET_PROPERTY:OBS::libobs,INTERFACE_INCLUDE_DIRECTORIES>)
target_link_libraries(studio-audio-roots-check PRIVATE "${STUDIO_AUDIO_TEST_CORE}" Threads::Threads ${CMAKE_DL_LIBS})
option(STUDIO_AUDIO_TEST_SHUTDOWN "Build private registry shutdown regression" OFF)
if(STUDIO_AUDIO_TEST_SHUTDOWN)
 add_executable(studio-audio-roots-shutdown-check shutdown-check.cpp)
 target_include_directories(studio-audio-roots-shutdown-check PRIVATE $<TARGET_PROPERTY:OBS::libobs,INTERFACE_INCLUDE_DIRECTORIES>)
 target_link_libraries(studio-audio-roots-shutdown-check PRIVATE "${STUDIO_AUDIO_TEST_CORE}" Threads::Threads ${CMAKE_DL_LIBS})
 if(MSVC)
  target_compile_options(studio-audio-roots-shutdown-check PRIVATE /utf-8)
  target_compile_definitions(studio-audio-roots-shutdown-check PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
 endif()
endif()

if(MSVC)
 target_compile_options(studio-audio-roots-check PRIVATE /utf-8)
 target_compile_definitions(studio-audio-roots-check PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
endif()
