cmake_minimum_required(VERSION 3.28)
project(studio-http-cancel LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(CURL REQUIRED)
find_package(Threads REQUIRED)
add_executable(studio-http-cancel-check check.cpp ../../frontend/oauth/OAuthHttpClient.cpp)
target_include_directories(studio-http-cancel-check PRIVATE ../../frontend)
target_link_libraries(studio-http-cancel-check PRIVATE CURL::libcurl Threads::Threads)
if(MSVC)
  target_compile_options(studio-http-cancel-check PRIVATE /utf-8)
endif()
