cmake_minimum_required(VERSION 3.28)
project(studio-interleave-keyframe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(STUDIO_INTERLEAVE_TEST_CORE "" CACHE FILEPATH "Private core with adapter; never installed")
if(NOT EXISTS "${STUDIO_INTERLEAVE_TEST_CORE}")
 message(FATAL_ERROR "Provide the exact private core built by prepare-private-core.py")
endif()
add_executable(studio-interleave-keyframe-check check.cpp)
target_link_libraries(studio-interleave-keyframe-check PRIVATE "${STUDIO_INTERLEAVE_TEST_CORE}")
if(MSVC)
 target_compile_options(studio-interleave-keyframe-check PRIVATE /utf-8)
endif()
